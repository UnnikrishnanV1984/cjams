CREATE OR REPLACE FUNCTION cjams.getuserworkloadcount_routingusers(v_toworkeridno character varying)
RETURNS TABLE(securityusersid character varying, casecount bigint, workload bigint) 
    LANGUAGE 'plpgsql'
    VOLATILE 
    COST 100
    ROWS 1000
AS $BODY$
BEGIN

RETURN QUERY
	SELECT 	toworkeridno securityusersid
			, COUNT(1) casecount
			, SUM(CASE WHEN enddate IS NULL THEN 1 ELSE 0 END)::bigint workload
	FROM	caseassignment  
	WHERE 	activeflag  = 1    
	and toworkeridno=v_toworkeridno
	GROUP BY toworkeridno;
	
END;
$BODY$;
